import { describe, expect, it } from 'vitest'
import { mountApp } from './main.js'

function fakeTransport() {
  let handlers = null
  const transport = (options) => {
    handlers = options
    return { close: () => undefined }
  }
  transport.push = (value) => handlers?.onFrame(JSON.parse(JSON.stringify(value)))
  return transport
}

function snapshotWithChoice() {
  return {
    kind: 'snapshot',
    sessionId: 's-replay',
    maxSeq: 3,
    frames: [
      { kind: 'turn', seq: 1, at: 1, turn: 1, phase: 'end', endKind: 'completed' },
      {
        kind: 'choice',
        seq: 2,
        at: 2,
        callId: 'call-ask',
        prompt: '要转给谁？',
        options: [{ id: 'a', label: '张伟' }],
        allowFreeText: true,
      },
      { kind: 'message', seq: 3, at: 3, role: 'user', blocks: [{ type: 'text', text: '张伟' }] },
    ],
  }
}

describe('probe · 快照重放对已答选择卡的影响', () => {
  it('答复后重连（同一会话全量快照）卡片会回到 open 可点状态', async () => {
    const root = document.createElement('div')
    document.body.append(root)
    const api = {
      session: async () => ({ ok: true, status: 200, data: { sessionId: 's-replay' } }),
      turn: async () => ({ ok: true, status: 202 }),
    }
    const transport = fakeTransport()
    mountApp({ root, api, transport })

    transport.push(snapshotWithChoice())
    const card = root.querySelector('[data-role="choice"]')
    card.querySelector('[data-role="choice-option"]').click()
    await Promise.resolve()
    await Promise.resolve()
    expect(card.dataset.state).toBe('picked')

    transport.push(snapshotWithChoice())
    const replayed = root.querySelector('[data-role="choice"]')
    expect(replayed).not.toBe(card)
    expect(replayed.dataset.state).toBe('open')
    expect(replayed.querySelectorAll('[data-role="choice-option"]').length).toBe(1)
  })
})
